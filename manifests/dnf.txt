# DNF packages: Fedora / RHEL (with EPEL for RHEL)
# For RHEL/CentOS enable EPEL first: sudo dnf install epel-release
#
# Install everything (dnf5) — sed strips full-line and inline comments:
#   sudo dnf5 install $(sed 's/#.*//' manifests/dnf.txt)
#   sudo dnf5 -y install $(sed 's/#.*//' manifests/dnf.txt)   # unattended
# Then provision neovim:
#   nvim --headless -c "luafile manifests/provision.lua" -c "qa!"
