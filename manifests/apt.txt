# APT packages — Ubuntu/Debian
# Add the unstable PPA first for a recent neovim build:
#   sudo add-apt-repository ppa:neovim-ppa/unstable
#   sudo apt update
#
# Install everything — sed strips full-line and inline comments:
#   sudo apt install -y $(sed 's/#.*//' manifests/apt.txt)
# Then provision neovim:
#   nvim --headless -c "luafile manifests/provision.lua" -c "qa!"
